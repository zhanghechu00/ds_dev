```python
class TaskSuccessEvaluator:
    """任务成功率评估器"""
    
    def __init__(self):
        self.total_tasks = 0
        self.successful_tasks = 0
        self.partial_success = 0
        self.failures = 0
    
    def evaluate_task(self, task: str, agent_output: str, 
                     ground_truth: str = None) -> dict:
        """评估单个任务"""
        
        self.total_tasks += 1
        
        # 1. 完成度评分
        completion_score = self.check_completion(task, agent_output)
        
        # 2. 准确性评分
        if ground_truth:
            accuracy_score = self.check_accuracy(agent_output, ground_truth)
        else:
            accuracy_score = None
        
        # 3. 判断成功等级
        if completion_score >= 0.9 and (accuracy_score is None or accuracy_score >= 0.9):
            self.successful_tasks += 1
            success_level = "完全成功"
        elif completion_score >= 0.6:
            self.partial_success += 1
            success_level = "部分成功"
        else:
            self.failures += 1
            success_level = "失败"
        
        return {
            "completion_score": completion_score,
            "accuracy_score": accuracy_score,
            "success_level": success_level
        }
    
    def get_metrics(self) -> dict:
        """获取整体指标"""
        return {
```

```python
class OutputQualityEvaluator:
    """多维度输出质量评估"""
    
    def __init__(self, llm):
        self.llm = llm
    
    def evaluate_quality(self, output: str, context: str = "") -> dict:
        """综合评估输出质量"""
        
        scores = {}
        
        # 1. 相关性 (Relevance)
        scores['relevance'] = self.evaluate_relevance(output, context)
        
        # 2. 连贯性 (Coherence)
        scores['coherence'] = self.evaluate_coherence(output)
        
        # 3. 流畅性 (Fluency)
        scores['fluency'] = self.evaluate_fluency(output)
        
        # 4. 准确性 (Accuracy)
        scores['accuracy'] = self.evaluate_accuracy(output, context)
        
        # 5. 信息量 (Informativeness)
        scores['informativeness'] = self.evaluate_informativeness(output)
        
        # 6. 安全性 (Safety)
        scores['safety'] = self.evaluate_safety(output)
        
        # 综合得分
        scores['overall'] = sum(scores.values()) / len(scores)
        
        return scores
    
    def evaluate_relevance(self, output: str, context: str) -> float:
        """评估相关性（0-1）"""
        prompt = f"""
        上下文：{context}
        输出：{output}
        
        评估输出与上下文的相关性（0-10分）。
        只返回数字。
        """
        score = float(self.llm.predict(prompt)) / 10
        return score
    
    def evaluate_coherence(self, output: str) -> float:
        """评估逻辑连贯性"""
        prompt = f"""
        文本：{output}
        
        评估此文本的逻辑连贯性（0-10分）。
        考虑：
        - 句子之间是否自然衔接
        - 论述是否有逻辑
        - 是否有矛盾之处
        
        只返回数字。
        """
        score = float(self.llm.predict(prompt)) / 10
        return score
    
    def evaluate_safety(self, output: str) -> float:
        """评估安全性"""
        # 检查有害内容
        harmful_keywords = ['暴力', '歧视', '非法', ...]
        
        if any(keyword in output for keyword in harmful_keywords):
            return 0.0
        
        # 使用安全分类器
        safety_score = self.safety_classifier.predict(output)
        return safety_score
```

```python
import time
from typing import Callable

class PerformanceMetrics:
    """性能和效率指标"""
    
    def __init__(self):
        self.measurements = []
    
    def measure_execution(self, agent_func: Callable, task: str) -> dict:
        """测量执行性能"""
        
        # 1. 延迟 (Latency)
        start_time = time.time()
        result = agent_func(task)
        end_time = time.time()
        latency = end_time - start_time
        
        # 2. Token使用量
        tokens_used = self.count_tokens(task, result)
        
        # 3. API调用次数
        api_calls = self.count_api_calls()
        
        # 4. 成本估算
        cost = self.estimate_cost(tokens_used, api_calls)
        
        metrics = {
            "latency_seconds": latency,
            "tokens_used": tokens_used,
            "api_calls": api_calls,
            "estimated_cost_usd": cost,
            "throughput": 1 / latency  # 每秒处理任务数
        }
        
        self.measurements.append(metrics)
        return metrics
    
    def get_statistics(self) -> dict:
        """获取统计数据"""
        if not self.measurements:
            return {}
        
        latencies = [m['latency_seconds'] for m in self.measurements]
        tokens = [m['tokens_used'] for m in self.measurements]
        costs = [m['estimated_cost_usd'] for m in self.measurements]
        
        return {
            "avg_latency": sum(latencies) / len(latencies),
            "p50_latency": self.percentile(latencies, 50),
            "p95_latency": self.percentile(latencies, 95),
            "p99_latency": self.percentile(latencies, 99),
            "total_tokens": sum(tokens),
            "avg_tokens": sum(tokens) / len(tokens),
            "total_cost": sum(costs),
            "requests_per_second": len(self.measurements) / sum(latencies)
        }
```

```python
class ReliabilityTester:
    """可靠性和鲁棒性测试"""
    
    def test_consistency(self, agent, task: str, runs: int = 10) -> dict:
        """测试输出一致性"""
        
        outputs = []
        for _ in range(runs):
            output = agent.run(task)
            outputs.append(output)
        
        # 计算输出的相似度
        similarities = []
        for i in range(len(outputs)):
            for j in range(i+1, len(outputs)):
                sim = self.compute_similarity(outputs[i], outputs[j])
                similarities.append(sim)
        
        avg_similarity = sum(similarities) / len(similarities)
        
        return {
            "consistency_score": avg_similarity,
            "variance": self.compute_variance(outputs),
            "unique_outputs": len(set(outputs))
        }
    
    def test_robustness(self, agent, task: str) -> dict:
        """测试对输入扰动的鲁棒性"""
        
        # 创建扰动版本
        perturbations = [
            task,  # 原始
            task + " ",  # 空格
            task.lower(),  # 小写
            self.add_typos(task),  # 拼写错误
            self.paraphrase(task)  # 换个说法
        ]
        
        outputs = [agent.run(p) for p in perturbations]
        
        # 检查输出是否仍然合理
        robustness_scores = []
        for output in outputs[1:]:  # 跟原始输出对比
            score = self.compute_similarity(outputs[0], output)
            robustness_scores.append(score)
        
        return {
            "robustness_score": sum(robustness_scores) / len(robustness_scores),
            "degradation": max(robustness_scores) - min(robustness_scores)
        }
    
    def test_error_handling(self, agent, edge_cases: list) -> dict:
        """测试异常处理能力"""
        
        results = []
        for case in edge_cases:
            try:
                output = agent.run(case)
                results.append({
                    "case": case,
                    "handled": True,
                    "output": output,
                    "graceful": self.is_graceful_response(output)
                })
            except Exception as e:
                results.append({
                    "case": case,
                    "handled": False,
                    "error": str(e),
                    "graceful": False
                })
        
        handled_count = sum(1 for r in results if r['handled'])
        graceful_count = sum(1 for r in results if r.get('graceful', False))
        
        return {
            "error_handling_rate": handled_count / len(edge_cases),
            "graceful_degradation_rate": graceful_count / len(edge_cases),
            "details": results
        }
```

```python
class UserExperienceEvaluator:
    """用户体验评估"""
    
    def __init__(self):
        self.ratings = []
    
    def collect_rating(self, task: str, output: str, user_feedback: dict):
        """收集用户评分"""
        
        rating = {
            "task": task,
            "output": output,
            "helpfulness": user_feedback.get("helpfulness", 0),  # 1-5
            "clarity": user_feedback.get("clarity", 0),  # 1-5
            "completeness": user_feedback.get("completeness", 0),  # 1-5
            "satisfaction": user_feedback.get("satisfaction", 0),  # 1-5
            "would_use_again": user_feedback.get("would_use_again", False),
            "comments": user_feedback.get("comments", "")
        }
        
        self.ratings.append(rating)
    
    def get_ux_metrics(self) -> dict:
        """计算用户体验指标"""
        
        if not self.ratings:
            return {}
        
        return {
            "avg_helpfulness": self.average("helpfulness"),
            "avg_clarity": self.average("clarity"),
            "avg_completeness": self.average("completeness"),
            "avg_satisfaction": self.average("satisfaction"),
            "net_promoter_score": self.calculate_nps(),
            "retention_rate": sum(
                1 for r in self.ratings if r['would_use_again']
            ) / len(self.ratings)
        }
    
    def calculate_nps(self) -> float:
        """计算净推荐值 (Net Promoter Score)"""
        # 基于satisfaction分数
        promoters = sum(1 for r in self.ratings if r['satisfaction'] >= 4)
        detractors = sum(1 for r in self.ratings if r['satisfaction'] <= 2)
        
        nps = ((promoters - detractors) / len(self.ratings)) * 100
        return nps
```

```python
class ComprehensiveEvaluator:
    """综合评估框架"""
    
    def __init__(self, llm):
        self.task_evaluator = TaskSuccessEvaluator()
        self.quality_evaluator = OutputQualityEvaluator(llm)
        self.performance_metrics = PerformanceMetrics()
        self.reliability_tester = ReliabilityTester()
        self.ux_evaluator = UserExperienceEvaluator()
    
    def evaluate_agent(self, agent, test_suite: dict) -> dict:
        """完整评估一个Agent"""
        
        results = {
            "task_success": {},
            "quality": {},
            "performance": {},
            "reliability": {},
            "user_experience": {}
        }
        
        # 1. 任务成功率
        for task in test_suite['tasks']:
            self.task_evaluator.evaluate_task(
                task['input'],
                agent.run(task['input']),
                task.get('expected_output')
            )
        results["task_success"] = self.task_evaluator.get_metrics()
        
        # 2. 输出质量
        quality_scores = []
        for task in test_suite['tasks']:
            output = agent.run(task['input'])
            quality = self.quality_evaluator.evaluate_quality(
                output, task['input']
            )
            quality_scores.append(quality)
        results["quality"] = self.aggregate_quality_scores(quality_scores)
        
        # 3. 性能指标
        for task in test_suite['tasks']:
            self.performance_metrics.measure_execution(
                agent.run, task['input']
            )
        results["performance"] = self.performance_metrics.get_statistics()
        
        # 4. 可靠性
        if test_suite.get('consistency_test'):
            results["reliability"]["consistency"] = \
                self.reliability_tester.test_consistency(
                    agent, test_suite['consistency_test']
                )
        
        if test_suite.get('robustness_test'):
            results["reliability"]["robustness"] = \
                self.reliability_tester.test_robustness(
                    agent, test_suite['robustness_test']
                )
        
        # 5. 用户体验（如果有人工评分）
        if test_suite.get('user_ratings'):
            results["user_experience"] = self.ux_evaluator.get_ux_metrics()
        
        # 6. 综合评分
        results["overall_score"] = self.calculate_overall_score(results)
        
        return results
    
    def calculate_overall_score(self, results: dict) -> float:
        """计算综合得分（0-100）"""
        
        weights = {
            "task_success": 0.35,
            "quality": 0.25,
            "performance": 0.15,
            "reliability": 0.15,
            "user_experience": 0.10
        }
        
        scores = {
            "task_success": results["task_success"]["success_rate"] * 100,
            "quality": results["quality"]["overall"] * 100,
            "performance": self.normalize_performance(results["performance"]),
            "reliability": results["reliability"].get("consistency", {}).get("consistency_score", 0.8) * 100,
            "user_experience": results.get("user_experience", {}).get("avg_satisfaction", 4) * 20
        }
        
        overall = sum(scores[k] * weights[k] for k in weights.keys())
        return overall
```

```python
        ax.set_xticklabels(metrics)
```

## 总结：关键评估指标

| 维度 | 核心指标 | 优先级 |
|------|---------|--------|
| **任务完成** | 成功率、准确率、F1分数 | 🔴 最高 |
| **输出质量** | 相关性、连贯性、准确性 | 🔴 最高 |
| **效率** | 延迟、Token使用、成本 | 🟡 高 |
| **可靠性** | 一致性、鲁棒性、错误处理 | 🟡 高 |
| **用户体验** | 满意度、NPS、留存率 | 🟢 中 |

**评估最佳实践：**

1. **建立基线**：先测量未优化版本的性能
2. **多维度评估**：不要只看单一指标
3. **统计显著性**：确保改进是真实的，不是随机波动
4. **持续监控**：上线后继续追踪指标变化
5. **用户反馈**：最终以实际用户体验为准

这套评估体系可以量化地证明优化的价值，为技术决策提供数据支持。
